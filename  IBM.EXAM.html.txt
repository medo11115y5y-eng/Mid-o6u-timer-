        // Load data from localStorage on start
        function loadSavedState() {
            try {
                const savedMistakes = localStorage.getItem('imb_dr_mohamed_mistakes');
                if (savedMistakes) {
                    state.userMistakesHistory = new Set(JSON.parse(savedMistakes));
                }
                const savedAnswers = localStorage.getItem('imb_dr_mohamed_answered');
                if (savedAnswers) {
                    state.answeredHistory = new Set(JSON.parse(savedAnswers));
                }
            } catch(e) {
                console.error("Local storage error:", e);
            }
        }

        function saveStateToStorage() {
            try {
                localStorage.setItem('imb_dr_mohamed_mistakes', JSON.stringify([...state.userMistakesHistory]));
                localStorage.setItem('imb_dr_mohamed_answered', JSON.stringify([...state.answeredHistory]));
            } catch(e) {
                console.error("Local storage save error:", e);
            }
        }

        function startQuiz(type, param) {
            state.activeQuizType = type;
            state.userAnswers = {};
            state.flaggedQuestions.clear();
            state.currentQuestionIndex = 0;
            state.elapsedSeconds = 0;

            if (type === 'full') {
                state.activeQuizQuestions = [...ALL_QUESTIONS];
                document.getElementById('quiz-title-display').innerText = "الميدتيرم الشامل (Full Mid-Term Exam)";
            } else if (type === 'lesson') {
                const lessonMeta = LESSONS_METADATA.find(l => l.id === param);
                state.activeQuizQuestions = ALL_QUESTIONS.filter(q => q.lessonId === param);
                document.getElementById('quiz-title-display').innerText = lessonMeta.title;
            } else if (type === 'random') {
                const count = param || 20;
                const shuffled = [...ALL_QUESTIONS].sort(() => 0.5 - Math.random());
                state.activeQuizQuestions = shuffled.slice(0, count);
                document.getElementById('quiz-title-display').innerText = `اختبار عشوائي (${count} سؤال)`;
            } else if (type === 'mistakes') {
                if (state.userMistakesHistory.size === 0) {
                    showModal("لا يوجد أسئلة خاطئة!", "ممتاز جداً دكتور محمد! ليس لديك أي أسئلة خاطئة مسجلة في بنك الأخطاء حالياً.", null);
                    return;
                }
                state.activeQuizQuestions = ALL_QUESTIONS.filter(q => state.userMistakesHistory.has(q.id));
                document.getElementById('quiz-title-display').innerText = `إعادة حل بنك الأخطاء (${state.activeQuizQuestions.length} سؤال)`;
            }

            // Switch Screen
            document.getElementById('screen-home').classList.add('hidden');
            document.getElementById('screen-results').classList.add('hidden');
            document.getElementById('screen-quiz').classList.remove('hidden');

            // Setup Timer
            setupTimer();

            // Render
            renderCurrentQuestion();
            renderNavigatorGrid();
        }

        function setupTimer() {
            clearInterval(state.timerInterval);
            const timerBadge = document.getElementById('timer-badge');
            const timerText = document.getElementById('timer-text');

            if (state.timerOption === 'none') {
                timerBadge.classList.add('hidden');
                timerBadge.classList.remove('flex');
            } else {
                timerBadge.classList.remove('hidden');
                timerBadge.classList.add('flex');
                
                if (state.timerOption === 'countup') {
                    state.elapsedSeconds = 0;
                    timerText.innerText = "00:00";
                    state.timerInterval = setInterval(() => {
                        state.elapsedSeconds++;
                        const mins = String(Math.floor(state.elapsedSeconds / 60)).padStart(2, '0');
                        const secs = String(state.elapsedSeconds % 60).padStart(2, '0');
                        timerText.innerText = `${mins}:${secs}`;
                    }, 1000);
                } else {
                    const durationMins = parseInt(state.timerOption);
                    state.remainingSeconds = durationMins * 60;
                    state.timerInterval = setInterval(() => {
                        state.remainingSeconds--;
                        state.elapsedSeconds++;
                        if (state.remainingSeconds <= 0) {
                            clearInterval(state.timerInterval);
                            submitQuiz();
                            showModal("انتهى الوقت!", "لقد انتهى وقت الاختبار المخصص وتم تصحيح الإجابات تلقائياً.", null);
                        }
                        const mins = String(Math.floor(state.remainingSeconds / 60)).padStart(2, '0');
                        const secs = String(state.remainingSeconds % 60).padStart(2, '0');
                        timerText.innerText = `${mins}:${secs}`;
                    }, 1000);
                }
            }
        }

            // Options
            const optsContainer = document.getElementById('options-container');
            optsContainer.innerHTML = '';

            const selectedOpt = state.userAnswers[q.id];
            const letters = ['a', 'b', 'c', 'd'];

            q.options.forEach((optText, idx) => {
                const isSelected = selectedOpt === idx;
                const optBtn = document.createElement('div');
                
                let extraClasses = "border-slate-200 hover:border-teal-400 hover:bg-slate-50";
                
                // In Practice mode, show correct/wrong immediately if answered

                    if (idx === q.answer) {
                        extraClasses = "border-emerald-500 bg-emerald-50 text-emerald-900 font-bold ring-1 ring-emerald-500";
                    } else if (isSelected && idx !== q.answer) {
                        extraClasses = "border-red-400 bg-red-50 text-red-900 ring-1 ring-red-400";

                } else if (isSelected) {
                    extraClasses = "border-teal-600 bg-teal-50 text-teal-900 font-bold ring-2 ring-teal-600";
                }

                optBtn.className = `en-text p-4 rounded-xl border text-sm font-medium transition-all cursor-pointer flex items-center justify-between ${extraClasses}`;

        function selectOption(qId, optIdx) {
            state.userAnswers[qId] = optIdx;
            state.answeredHistory.add(qId);
            saveStateToStorage();

            renderCurrentQuestion();
            renderNavigatorGrid();
        }

        function clearCurrentAnswer() {
            const q = state.activeQuizQuestions[state.currentQuestionIndex];
            delete state.userAnswers[q.id];
            renderCurrentQuestion();
            renderNavigatorGrid();
        }

        function toggleFlagCurrentQuestion() {
            const q = state.activeQuizQuestions[state.currentQuestionIndex];
            if (state.flaggedQuestions.has(q.id)) {
                state.flaggedQuestions.delete(q.id);
            } else {
                state.flaggedQuestions.add(q.id);
            }
            renderCurrentQuestion();
            renderNavigatorGrid();
        }

        function prevQuestion() {
            if (state.currentQuestionIndex > 0) {
                state.currentQuestionIndex--;
                renderCurrentQuestion();
                renderNavigatorGrid();
            }
        }

        function nextQuestion() {
            if (state.currentQuestionIndex < state.activeQuizQuestions.length - 1) {
                state.currentQuestionIndex++;
                renderCurrentQuestion();
                renderNavigatorGrid();
            }
        }

        /* ==========================================================================
           5. NAVIGATOR GRID & FILTERS
           ========================================================================== */
        function renderNavigatorGrid() {
            const grid = document.getElementById('navigator-grid');
            grid.innerHTML = '';

            const total = state.activeQuizQuestions.length;
            let answeredCount = 0;

            state.activeQuizQuestions.forEach((q, idx) => {
                const isAnswered = state.userAnswers[q.id] !== undefined;
                const isFlagged = state.flaggedQuestions.has(q.id);
                if (isAnswered) answeredCount++;

                // Filter check
                if (state.activeFilterGrid === 'flagged' && !isFlagged) return;
                if (state.activeFilterGrid === 'unanswered' && isAnswered) return;

                const btn = document.createElement('button');
                btn.className = `w-full h-9 rounded-lg font-bold text-xs flex items-center justify-center transition-all relative ${
                    state.currentQuestionIndex === idx ? 'ring-2 ring-slate-900 ring-offset-1' : ''
                } ${
                    isFlagged ? 'bg-amber-400 text-slate-900 font-extrabold' : 
                    (isAnswered ? 'bg-medical-700 text-white' : 'bg-slate-100 text-slate-600 hover:bg-slate-200')
                }`;

                btn.innerText = idx + 1;
                btn.onclick = () => {
                    state.currentQuestionIndex = idx;
                    renderCurrentQuestion();
                    renderNavigatorGrid();
                };

                grid.appendChild(btn);
            });

            document.getElementById('nav-counter').innerText = `${answeredCount}/${total}`;
        }

        function filterGrid(filterType) {

        function submitQuiz() {
            clearInterval(state.timerInterval);

            // Calculate Score & Update Mistakes History
            let correctCount = 0;
            let wrongCount = 0;
            let skippedCount = 0;

            state.activeQuizQuestions.forEach(q => {
                const userAns = state.userAnswers[q.id];
                if (userAns === undefined) {
                    skippedCount++;
                } else if (userAns === q.answer) {
                    correctCount++;
                    state.userMistakesHistory.delete(q.id); // Remove from mistakes if corrected
                } else {
                    wrongCount++;
                    state.userMistakesHistory.add(q.id); // Save to mistakes bank
                }
            });

            saveStateToStorage();
            updateHomeStats();

            // Switch Screen
            document.getElementById('screen-quiz').classList.add('hidden');
            document.getElementById('screen-results').classList.remove('hidden');

            // Render Results Data
            const total = state.activeQuizQuestions.length;
            const percent = Math.round((correctCount / total) * 100);

            document.getElementById('result-total-qs').innerText = total;
            document.getElementById('result-correct-qs').innerText = correctCount;
            document.getElementById('result-wrong-qs').innerText = wrongCount;
            document.getElementById('result-skipped-qs').innerText = skippedCount;

            document.getElementById('result-score-percent').innerText = `${percent}%`;

            // SVG Circle Progress
            const svgCircle = document.getElementById('result-svg-circle');
            svgCircle.setAttribute('stroke-dasharray', `${percent}, 100`);

            // Grade Badge & Feedback
            const gradeBadge = document.getElementById('result-grade-badge');
            const feedbackText = document.getElementById('result-feedback-text');

            if (percent >= 90) {
                gradeBadge.className = "text-xs font-bold px-2.5 py-0.5 rounded bg-emerald-100 text-emerald-800 mt-1";
                gradeBadge.innerText = "امتياز ممتاز 🌟";
                feedbackText.innerText = "أداء استثنائي يا دكتور محمد! إتقان تام لمفاهيم وموديول الـ IMB، استمر بهذا المستوى العالي!";
            } else if (percent >= 75) {
                gradeBadge.className = "text-xs font-bold px-2.5 py-0.5 rounded bg-teal-100 text-teal-800 mt-1";
                gradeBadge.innerText = "جيد جداً مرتفع 👍";
                feedbackText.innerText = "عمل رائع جداً! استعراض ممتاز للأسئلة، راجع الأخطاء البسيطة لتقفل الدرجة النهائية.";
            } else if (percent >= 60) {
                gradeBadge.className = "text-xs font-bold px-2.5 py-0.5 rounded bg-amber-100 text-amber-800 mt-1";
                gradeBadge.innerText = "جيد - يحتاج تحسين";
                feedbackText.innerText = "نتيجة مقبولة، لكن يُفضل إعادة مراجعة الدروس ذات النسب المنخفضة من بنك الأخطاء لتضمن النتيجة.";
            } else {
                gradeBadge.className = "text-xs font-bold px-2.5 py-0.5 rounded bg-red-100 text-red-800 mt-1";
                gradeBadge.innerText = "يحتاج إعادة مذاكرة";
                feedbackText.innerText = "لا تقلق يا دكتور، اعتبر هذا الاختبار خطوة أولى لتحديد النقاط الضعيفة، قم بمراجعة التوضيحات العلمية وأعد الحل.";
            }

            // Render Lesson Breakdown & Detailed Review List
            renderLessonPerformanceBreakdown();
            renderReviewList();
        }

        function renderLessonPerformanceBreakdown() {
            const container = document.getElementById('lesson-performance-breakdown');
            container.innerHTML = '';

            LESSONS_METADATA.forEach(lesson => {
                const lessonQsInQuiz = state.activeQuizQuestions.filter(q => q.lessonId === lesson.id);
                if (lessonQsInQuiz.length === 0) return;

                let correctInLesson = 0;
                lessonQsInQuiz.forEach(q => {
                    if (state.userAnswers[q.id] === q.answer) correctInLesson++;
                });

                const pct = Math.round((correctInLesson / lessonQsInQuiz.length) * 100);

                const item = document.createElement('div');
                item.className = "flex items-center justify-between gap-4 text-xs font-semibold";
                item.innerHTML = `
                    <div class="w-1/3 truncate text-slate-700 font-bold">${lesson.title}</div>
                    <div class="w-1/2 bg-slate-100 h-2 rounded-full overflow-hidden">
                        <div class="bg-teal-600 h-full transition-all" style="width: ${pct}%"></div>
                    </div>
                    <div class="w-12 text-left text-slate-800 font-bold">${pct}%</div>
                `;
                container.appendChild(item);
            });
        }

        function renderReviewList() {
            const container = document.getElementById('review-items-container');
            container.innerHTML = '';

            const letters = ['a', 'b', 'c', 'd'];

            state.activeQuizQuestions.forEach((q, idx) => {
                const userAns = state.userAnswers[q.id];
                const isCorrect = userAns === q.answer;
                const isSkipped = userAns === undefined;
                const isFlagged = state.flaggedQuestions.has(q.id);

                // Filter check
                if (state.activeReviewFilter === 'correct' && !isCorrect) return;
                if (state.activeReviewFilter === 'incorrect' && (isCorrect || isSkipped)) return;
                if (state.activeReviewFilter === 'flagged' && !isFlagged) return;

                const card = document.createElement('div');
                card.className = `p-5 rounded-xl border text-xs sm:text-sm ${
                    isCorrect ? 'bg-emerald-50/40 border-emerald-200' : 
                    (isSkipped ? 'bg-amber-50/40 border-amber-200' : 'bg-red-50/40 border-red-200')
                }`;

                let optsMarkup = '';
                q.options.forEach((optText, oIdx) => {
                    let optStyle = "border-slate-200 bg-white text-slate-700";
                    if (oIdx === q.answer) {
                        optStyle = "border-emerald-500 bg-emerald-100 text-emerald-900 font-bold";
                    } else if (userAns === oIdx && oIdx !== q.answer) {
                        optStyle = "border-red-400 bg-red-100 text-red-900 font-bold";
                    }

                    optsMarkup += `
                        <div class="en-text p-2.5 rounded-lg border flex items-center justify-between ${optStyle}">
                            <span><strong class="uppercase mr-1 font-bold">${letters[oIdx]}.</strong> ${optText}</span>
                            ${oIdx === q.answer ? '<i class="fa-solid fa-check text-emerald-700"></i>' : 
                            (userAns === oIdx ? '<i class="fa-solid fa-xmark text-red-600"></i>' : '')}
                        </div>
                    `;
                });

                card.innerHTML = `
                    <div class="flex items-center justify-between gap-2 mb-2 pb-2 border-b border-slate-200/60">
                        <span class="font-bold text-slate-700">سؤال رقم ${idx + 1} (Orig. #${q.orig})</span>
                        <div class="flex items-center gap-2">
                            ${isFlagged ? '<span class="px-2 py-0.5 rounded bg-amber-100 text-amber-800 text-[11px] font-bold">معلّم</span>' : ''}
                            <span class="px-2.5 py-0.5 rounded-full font-bold text-[11px] ${
                                isCorrect ? 'bg-emerald-100 text-emerald-800' : 
                                (isSkipped ? 'bg-amber-100 text-amber-800' : 'bg-red-100 text-red-800')
                            }">
                                ${isCorrect ? 'إجابة صحيحة' : (isSkipped ? 'تم تخطيه' : 'إجابة خاطئة')}
                            </span>
                        </div>
                    </div>
                    
                    <p class="en-text font-bold text-slate-900 text-base mb-4 leading-relaxed">${q.q}</p>
                    
                    <div class="space-y-2 mb-4">
                        ${optsMarkup}
                    </div>

                    <div class="p-3 bg-white rounded-lg border border-slate-200/80 text-slate-700 leading-relaxed text-xs">
                        <strong class="text-medical-700 block mb-1 font-bold"><i class="fa-solid fa-lightbulb"></i> الشرح والتوضيح:</strong>
                        ${q.exp}
                    </div>
                `;

                container.appendChild(card);
            });
        }

        function filterReviewList(filter) {
            state.activeReviewFilter = filter;
            
            ['rf-all', 'rf-correct', 'rf-incorrect', 'rf-flagged'].forEach(id => {
                document.getElementById(id).className = "px-3 py-1 rounded-md text-slate-600 hover:bg-white/50";
            });
            document.getElementById(`rf-${filter}`).className = "px-3 py-1 rounded-md bg-white text-slate-900 shadow-sm font-bold";

            renderReviewList();
        }

        function showModal(title, bodyText, confirmCallback) {
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-body').innerText = bodyText;

            const confirmBtn = document.getElementById('modal-confirm-btn');
            if (confirmCallback) {
                confirmBtn.classList.remove('hidden');
                confirmBtn.onclick = () => {
                    confirmCallback();
                    closeModal();
                };
            } else {
                confirmBtn.classList.add('hidden');
            }

            document.getElementById('modal-container').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('modal-container').classList.add('hidden');
        }

                localStorage.removeItem('imb_dr_mohamed_mistakes');
                localStorage.removeItem('imb_dr_mohamed_answered');
                state.userMistakesHistory.clear();
                state.answeredHistory.clear();
                returnToHome();
            });
        }
